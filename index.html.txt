    <script>
        // 실시간 시계 및 날짜 업데이트
        function updateDateTime() {
            const now = new Date();
            const hours = String(now.getHours()).padStart(2, '0');
            const minutes = String(now.getMinutes()).padStart(2, '0');
            const timeStr = `${hours}:${minutes}`;
            
            document.getElementById('clock-small').innerText = timeStr;
            document.getElementById('clock-large').innerText = timeStr;

            // 날짜 표시 (예: 2026년 4월 20일 월요일)
            const year = now.getFullYear();
            const month = now.getMonth() + 1;
            const date = now.getDate();
            const days = ['일', '월', '화', '수', '목', '금', '토'];
            const dayName = days[now.getDay()];
            
            document.getElementById('current-date').innerText = `${year}년 ${month}월 ${date}일 ${dayName}요일`;
        }
        
        setInterval(updateDateTime, 1000);
        updateDateTime();
    </script>
